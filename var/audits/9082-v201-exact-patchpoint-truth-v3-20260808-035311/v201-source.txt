   258	def attach_v201_governing_inputs(
   259	    payload: Any,
   260	    *,
   261	    symbol: Any = None,
   262	    timeframe: Any = None,
   263	    analysis_mode: Any = None,
   264	    max_cot_age_days: int = DEFAULT_MAX_COT_AGE_DAYS,
   265	    raw_cot_override: Mapping[str, Any] | None = None,
   266	) -> Any:
   267	    if not isinstance(payload, dict):
   268	        return payload
   269	
   270	    data = deepcopy(payload)
   271	    instrument = data.get("instrument") if isinstance(data.get("instrument"), Mapping) else {}
   272	    normalized_symbol = str(symbol or instrument.get("symbol") or data.get("symbol") or "").upper().strip()
   273	    normalized_tf = str(timeframe or instrument.get("timeframe") or "weekly").strip()
   274	    mode = _analysis_mode(analysis_mode, normalized_tf)
   275	    raw = dict(raw_cot_override) if isinstance(raw_cot_override, Mapping) else _fetch_raw_cot(normalized_symbol)
   276	
   277	    directions = {
   278	        "asset_managers_overall": _normalize_direction(
   279	            raw.get("asset_managers_overall_direction") or raw.get("asset_managers_bias")
   280	        ),
   281	        "asset_managers_weekly": _normalize_direction(raw.get("asset_managers_weekly_direction")),
   282	        "leveraged_funds_overall": _normalize_direction(
   283	            raw.get("leveraged_funds_overall_direction")
   284	            or raw.get("leveraged_funds_bias")
   285	        ),
   286	        "leveraged_funds_weekly": _normalize_direction(raw.get("leveraged_funds_weekly_direction")),
   287	    }
   288	    missing = [name for name, value in directions.items() if value not in {"bullish", "bearish", "neutral"}]
   289	    age = _safe_float(raw.get("report_age_days"))
   290	
   291	    # Canonical freshness fallback:
   292	    # report_age_days may be derived only from a real report_date.
   293	    # Missing report_date remains missing; no synthetic freshness.
   294	    if age is None:
   295	        report_date_value = raw.get("report_date")
   296	        if report_date_value:
   297	            try:
   298	                from datetime import date, datetime, timezone
   299	
   300	                text = str(report_date_value).strip()
   301	                parsed = datetime.fromisoformat(
   302	                    text.replace("Z", "+00:00")
   303	                )
   304	
   305	                if parsed.tzinfo is None:
   306	                    parsed = parsed.replace(tzinfo=timezone.utc)
   307	
   308	                age = float(
   309	                    (datetime.now(timezone.utc).date() - parsed.date()).days
   310	                )
   311	            except Exception:
   312	                try:
   313	                    parsed_date = date.fromisoformat(
   314	                        str(report_date_value).strip()[:10]
   315	                    )
   316	                    age = float(
   317	                        (datetime.now(timezone.utc).date() - parsed_date).days
   318	                    )
   319	                except Exception:
   320	                    age = None
   321	
   322	    fresh = age is not None and age <= float(max_cot_age_days)
   323	    complete = (
   324	        raw.get("raw_cot_connected") is True
   325	        and raw.get("raw_cot_governing_inputs_complete") is True
   326	        and not missing
   327	        and fresh
   328	    )
   329	
   330	    data["governing_inputs"] = {
   331	        "contract_version": GOVERNING_INPUTS_VERSION,
   332	        "status": "COMPLETE" if complete else "INCOMPLETE",
   333	        "source": "CFTC_TFF_FUTURES_ONLY",
   334	        "source_family": raw.get("source_family"),
   335	        "source_market": raw.get("market_name"),
   336	        "report_date": raw.get("report_date"),
   337	        "report_age_days": age,
   338	        "max_age_days": int(max_cot_age_days),
   339	        "fresh": fresh,
   340	        "symbol": normalized_symbol,
   341	        "source_timeframe": "weekly",
   342	        "analysis_timeframe": normalized_tf,
   343	        "analysis_mode": mode,
   344	        "directions": directions,
   345	        "missing_inputs": missing,
   346	        "raw_cot_connected": raw.get("raw_cot_connected") is True,
   347	        "raw_cot_status": raw.get("raw_cot_status"),
   348	        "not_recommendation": True,
   349	    }
   350	
   351	    if not complete:
   352	        data["governing_inputs_reason_code"] = (
   353	            "COT_STALE" if age is not None and not fresh
   354	            else "COT_GOVERNING_INPUTS_INCOMPLETE"
   355	        )
   356	        return data
   357	
   358	    data.update(directions)
   359	    data["asset_managers_overall_direction"] = directions["asset_managers_overall"]
   360	    data["asset_managers_weekly_direction"] = directions["asset_managers_weekly"]
   361	    data["leveraged_funds_overall_direction"] = directions["leveraged_funds_overall"]
   362	    data["leveraged_funds_weekly_direction"] = directions["leveraged_funds_weekly"]
   363	    data["canonical_cot_directions"] = dict(directions)
   364	
   365	    data = apply_canonical_golden(data)
   366	    golden = data.get("golden_alignment")
   367	    if isinstance(golden, dict):
   368	        golden.update({
   369	            "governing_inputs_complete": True,
   370	            "governing_inputs_contract_version": GOVERNING_INPUTS_VERSION,
   371	            "source_report_date": raw.get("report_date"),
   372	            "source_report_age_days": age,
   373	            "source_fresh": fresh,
   374	            "source_family": raw.get("source_family"),
   375	            "decision_authority": False,
   376	            "not_recommendation": True,
   377	            "no_buy_sell": True,
   378	        })
   379	
   380	    _update_public_layers(data, mode, directions)
   381	    _refresh_public_golden(data)
   382	    data["governing_inputs_reason_code"] = None
   383	    data["governing_inputs_contract_version"] = GOVERNING_INPUTS_VERSION
   384	    return data
