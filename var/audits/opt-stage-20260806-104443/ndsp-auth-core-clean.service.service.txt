# /etc/systemd/system/ndsp-auth-core-clean.service
[Unit]
Description=NDSP Clean Authentication Core
After=network-online.target postgresql.service
Wants=network-online.target

[Service]
Type=simple
User=ndsp-auth
Group=ndsp-auth
WorkingDirectory=/opt/ndsp-auth-core-clean/current
EnvironmentFile=/etc/ndsp/ndsp-auth-core-clean.env
ExecStart=/usr/bin/nsolid /opt/ndsp-auth-core-clean/current/server/dist/server.js
Restart=on-failure
RestartSec=3
TimeoutStartSec=30
TimeoutStopSec=20
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=strict
ProtectHome=true
ReadWritePaths=/var/lib/ndsp-auth-core
CapabilityBoundingSet=
RestrictAddressFamilies=AF_UNIX AF_INET AF_INET6
LockPersonality=true

[Install]
WantedBy=multi-user.target
