# /etc/systemd/system/ndsp-v3-portal-gateway.service
[Unit]
Description=NDSP V3 Portal Gateway
After=network.target

[Service]
Type=simple
User=www-data
Group=www-data
WorkingDirectory=/opt/ndsp-v3-portal-gateway
ExecStart=/usr/bin/python3 /opt/ndsp-v3-portal-gateway/app.py
Restart=always
RestartSec=3

[Install]
WantedBy=multi-user.target
