# /etc/systemd/system/ndsp-ui-bridge-api.service
[Unit]
Description=NDSP UI Bridge API
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/ndsp-ui-bridge-api
ExecStart=/usr/bin/python3 -m uvicorn main:app --host 127.0.0.1 --port 9066
Restart=always
RestartSec=3

[Install]
WantedBy=multi-user.target
