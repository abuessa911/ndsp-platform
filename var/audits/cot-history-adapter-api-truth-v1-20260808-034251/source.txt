     1	from __future__ import annotations
     2	
     3	import csv
     4	from dataclasses import dataclass
     5	from datetime import date, datetime
     6	from pathlib import Path

    10	@dataclass(frozen=True)
    11	class CotMarketIdentity:
    12	    market_name: str
    13	    market_code: str
    14	    exchange: str
    15	
    16	
    17	@dataclass(frozen=True)
    18	class CotPositionSnapshot:
    19	    identity: CotMarketIdentity
    20	    report_date: date
    21	
    22	    asset_managers_long: int
    23	    asset_managers_short: int
    24	
    25	    leveraged_funds_long: int
    26	    leveraged_funds_short: int
    27	
    28	    source_file: str
    29	
    30	
    31	@dataclass(frozen=True)
    32	class CotWeeklyChange:
    33	    current_report_date: date
    34	    previous_report_date: date
    35	
    36	    asset_managers_change_long: int
    37	    asset_managers_change_short: int
    38	
    39	    leveraged_funds_change_long: int
    40	    leveraged_funds_change_short: int
    41	
    42	    source: str = "REAL_SAME_MARKET_CONSECUTIVE_CFTC_REPORTS"
    43	    synthetic_values_used: bool = False
    44	
    45	
    46	class CotHistoryError(RuntimeError):
    47	    pass

    50	def _int(value: str) -> int:
    51	    return int(str(value).replace(",", "").strip())
    52	
    53	
    54	def _date(value: str) -> date:
    55	    return datetime.strptime(
    56	        str(value).strip(),
    57	        "%Y-%m-%d",
    58	    ).date()
    59	
    60	
    61	def _read_snapshot(
    62	    path: Path,
    63	    target: CotMarketIdentity,
    64	) -> Optional[CotPositionSnapshot]:
    65	
    66	    with path.open(
    67	        "r",
    68	        errors="replace",
    69	        newline="",
    70	    ) as fh:
    71	
    72	        reader = csv.reader(fh)
    73	
    74	        for row in reader:
    75	
    76	            if len(row) <= 15:
    77	                continue
    78	
    79	            market_name = str(row[0]).strip()
    80	            market_code = str(row[3]).strip()
    81	            exchange = str(row[4]).strip()
    82	
    83	            if market_name != target.market_name:
    84	                continue
    85	
    86	            if market_code != target.market_code:
    87	                continue
    88	
    89	            if exchange != target.exchange:
    90	                continue
    91	
    92	            try:
    93	                report_date = _date(row[2])
    94	
    95	                return CotPositionSnapshot(
    96	                    identity=target,
    97	                    report_date=report_date,
    98	
    99	                    asset_managers_long=_int(row[11]),
   100	                    asset_managers_short=_int(row[12]),
   101	
   102	                    leveraged_funds_long=_int(row[14]),
   103	                    leveraged_funds_short=_int(row[15]),
   104	
   105	                    source_file=str(path),
   106	                )
   107	
   108	            except (ValueError, IndexError):
   109	                continue
   110	
   111	    return None

   114	def discover_same_market_history(
   115	    raw_dir: Path,
   116	    target: CotMarketIdentity,
   117	) -> list[CotPositionSnapshot]:
   118	
   119	    if not raw_dir.exists():
   120	        raise CotHistoryError(
   121	            "RAW_COT_DIRECTORY_NOT_FOUND"
   122	        )
   123	
   124	    candidates = sorted(
   125	        path
   126	        for path in raw_dir.glob("*FinFutWk*.txt")
   127	        if path.is_file()
   128	    )
   129	
   130	    snapshots: list[CotPositionSnapshot] = []
   131	
   132	    for path in candidates:
   133	
   134	        snapshot = _read_snapshot(
   135	            path,
   136	            target,
   137	        )
   138	
   139	        if snapshot is not None:
   140	            snapshots.append(snapshot)
   141	
   142	    # One canonical record for each distinct report date.
   143	    by_date: dict[date, CotPositionSnapshot] = {}
   144	
   145	    for snapshot in snapshots:
   146	
   147	        existing = by_date.get(
   148	            snapshot.report_date
   149	        )
   150	
   151	        if existing is None:
   152	            by_date[snapshot.report_date] = snapshot
   153	            continue
   154	
   155	        current_name = Path(
   156	            snapshot.source_file
   157	        ).name
   158	
   159	        existing_name = Path(
   160	            existing.source_file
   161	        ).name
   162	
   163	        # If the same report exists in both an archived file
   164	        # and the canonical current file, prefer the current file.
   165	        if (
   166	            current_name
   167	            == "current_tff_futures_only_FinFutWk.txt"
   168	            and existing_name != current_name
   169	        ):
   170	            by_date[snapshot.report_date] = snapshot
   171	
   172	    history = [
   173	        by_date[report_date]
   174	        for report_date in sorted(
   175	            by_date.keys(),
   176	            reverse=True,
   177	        )
   178	    ]
   179	
   180	    return history

   183	def derive_weekly_change(
   184	    history: list[CotPositionSnapshot],
   185	) -> CotWeeklyChange:
   186	
   187	    if len(history) < 2:
   188	        raise CotHistoryError(
   189	            "NO_TWO_DISTINCT_REPORT_DATES_FOR_EXACT_SAME_MARKET"
   190	        )
   191	
   192	    current = history[0]
   193	    previous = history[1]
   194	
   195	    if current.identity != previous.identity:
   196	        raise CotHistoryError(
   197	            "MARKET_IDENTITY_MISMATCH"
   198	        )
   199	
   200	    if current.report_date == previous.report_date:
   201	        raise CotHistoryError(
   202	            "REPORT_DATES_NOT_DISTINCT"
   203	        )
   204	
   205	    return CotWeeklyChange(
   206	        current_report_date=current.report_date,
   207	        previous_report_date=previous.report_date,
   208	
   209	        asset_managers_change_long=(
   210	            current.asset_managers_long
   211	            - previous.asset_managers_long
   212	        ),
   213	
   214	        asset_managers_change_short=(
   215	            current.asset_managers_short
   216	            - previous.asset_managers_short
   217	        ),
   218	
   219	        leveraged_funds_change_long=(
   220	            current.leveraged_funds_long
   221	            - previous.leveraged_funds_long
   222	        ),
   223	
   224	        leveraged_funds_change_short=(
   225	            current.leveraged_funds_short
   226	            - previous.leveraged_funds_short
   227	        ),
   228	    )

   231	def build_governed_history_payload(
   232	    raw_dir: Path,
   233	    target: CotMarketIdentity,
   234	) -> dict:
   235	
   236	    history = discover_same_market_history(
   237	        raw_dir,
   238	        target,
   239	    )
   240	
   241	    if not history:
   242	        raise CotHistoryError(
   243	            "NO_EXACT_MARKET_HISTORY"
   244	        )
   245	
   246	    current = history[0]
   247	
   248	    weekly = derive_weekly_change(
   249	        history
   250	    )
   251	
   252	    return {
   253	        "market_name": target.market_name,
   254	        "market_code": target.market_code,
   255	        "exchange": target.exchange,
   256	
   257	        "report_date": str(
   258	            current.report_date
   259	        ),
   260	
   261	        "previous_report_date": str(
   262	            weekly.previous_report_date
   263	        ),
   264	
   265	        "asset_managers_long":
   266	            current.asset_managers_long,
   267	
   268	        "asset_managers_short":
   269	            current.asset_managers_short,
   270	
   271	        "asset_managers_change_long":
   272	            weekly.asset_managers_change_long,
   273	
   274	        "asset_managers_change_short":
   275	            weekly.asset_managers_change_short,
   276	
   277	        "leveraged_funds_long":
   278	            current.leveraged_funds_long,
   279	
   280	        "leveraged_funds_short":
   281	            current.leveraged_funds_short,
   282	
   283	        "leveraged_funds_change_long":
   284	            weekly.leveraged_funds_change_long,
   285	
   286	        "leveraged_funds_change_short":
   287	            weekly.leveraged_funds_change_short,
   288	
   289	        "weekly_change_source":
   290	            weekly.source,
   291	
   292	        "synthetic_values_used":
   293	            weekly.synthetic_values_used,
   294	
   295	        "history_depth":
   296	            len(history),
   297	
   298	        "history_dates": [
   299	            str(snapshot.report_date)
   300	            for snapshot in history
   301	        ],
   302	
   303	        "current_source_file":
   304	            current.source_file,
   305	    }
